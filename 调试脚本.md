// ===== 1. 检查当前身份 =====
console.log('========== 当前身份 ==========')
const currentRole = wx.getStorageSync('currentRole')
console.log('当前身份:', currentRole)
console.log('是否为赛事主办方:', currentRole === 'organizer')
console.log('')

// ===== 2. 检查球队数据 =====
console.log('========== 球队数据 ==========')
const db = wx.cloud.database()

async function checkTeamData() {
  // 查询所有球队
  const allTeams = await db.collection('tournament_teams').get()
  console.log('所有球队数量:', allTeams.data.length)

  // 查询待审核球队
  const pendingTeams = await db.collection('tournament_teams')
    .where({ status: 'pending' })
    .get()
  console.log('待审核球队:', pendingTeams.data)
  console.log('待审核数量:', pendingTeams.data.length)

  // 查询已通过球队
  const approvedTeams = await db.collection('tournament_teams')
    .where({ status: 'approved' })
    .get()
  console.log('已通过球队:', approvedTeams.data)
  console.log('已通过数量:', approvedTeams.data.length)
}

checkTeamData()
console.log('')
